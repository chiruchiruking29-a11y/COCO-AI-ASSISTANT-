<script>﻿

const reply = document.getElementById("reply");﻿

const statusText = document.getElementById("status");﻿

function answer(text) {﻿

  const q = text.toLowerCase();﻿

  let a = "I'm COCO! This is my demo version. AI connection will be added next.";﻿

  if (q.includes("hello") || q.includes("hi"))﻿

    a = "Hello macha! I'm COCO. How can I help you?";﻿

  else if (q.includes("your name"))﻿

    a = "My name is COCO, your personal assistant.";﻿

  else if (q.includes("time"))﻿

    a = "The time is " + new Date().toLocaleTimeString();﻿

  else if (q.includes("date"))﻿

    a = "Today's date is " + new Date().toLocaleDateString();﻿

  else if (q.includes("how are you"))﻿

    a = "I'm ready to help you!";﻿

﻿

  reply.textContent = a;﻿

  statusText.textContent = "Reply ready";﻿

  speechSynthesis.cancel();﻿

  speechSynthesis.speak(new SpeechSynthesisUtterance(a));﻿

}﻿

function send() {﻿

  const input = document.getElementById("msg");﻿

  if (!input.value.trim()) return;﻿

  answer(input.value);﻿

  input.value = "";﻿

}﻿

function listen() {﻿

  const SR = window.SpeechRecognition ||﻿

             window.webkitSpeechRecognition;﻿

  if (!SR) {﻿

    statusText.textContent =﻿

      "Voice recognition unavailable. Try Chrome or type instead.";﻿

    return;﻿

  }﻿

  const rec = new SR();﻿

  rec.lang = "en-IN";﻿

  rec.onstart = () => statusText.textContent = "Listening...";﻿

  rec.onresult = e => {﻿

    document.getElementById("msg").value =﻿

      e.results[0][0].transcript;﻿

    send();﻿

  };﻿

  rec.onerror = () => statusText.textContent =﻿

    "Microphone issue. Allow microphone access and retry.";﻿

  rec.onend = () => {﻿

    if (statusText.textContent === "Listening...")﻿

      statusText.textContent = "Ready when you are";﻿

  };﻿

  rec.start();﻿

}﻿

</script>﻿
